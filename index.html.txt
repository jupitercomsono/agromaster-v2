function getDB(){
  const db = localStorage.getItem('agromaster_db');

function cadastrar(){
  const nome=document.getElementById('cad-nome').value.trim();
  const email=document.getElementById('cad-email').value.trim();
  const whats=document.getElementById('cad-whats').value.trim();
  const senha=document.getElementById('cad-senha').value;
  const senha2=document.getElementById('cad-senha2').value;

  // limpa erros
  document.querySelectorAll('.input-error').forEach(e=>e.classList.remove('show'));
  let ok=true;
  if(!nome){document.getElementById('err-nome').classList.add('show');ok=false;}
  if(!email||!email.includes('@')){document.getElementById('err-email').classList.add('show');ok=false;}
  if(!whats){document.getElementById('err-whats').classList.add('show');ok=false;}
  if(senha.length<6){document.getElementById('err-senha').classList.add('show');ok=false;}
  if(senha!==senha2){document.getElementById('err-senha2').classList.add('show');ok=false;}
  if(!ok) return;

  const db=getDB();
  if(db.users.find(u=>u.email===email)){
    const al=document.getElementById('cad-alert');
    al.textContent='Este email já está cadastrado. Faça login!';
    al.className='alert error show';return;
  }

  cadData={nome,email,whats,senha};
  // Atualiza PIX display
  document.getElementById('pix-key-display').textContent=db.config.pixKey||'Chave não configurada';
  showScreen('pagamento');
}

function login(){
  const email=document.getElementById('login-email').value.trim();
  const senha=document.getElementById('login-senha').value;
  const al=document.getElementById('login-alert');

  if(!email||!senha){
    al.textContent='Preencha email e senha';al.className='alert error show';return;
  }

  // Admin login
  if(email==='admin@agromaster.com'&&senha==='admin123'){
    setSession('admin');
    loadAdmin();
    showScreen('admin');return;
  }

  const db=getDB();
  const user=db.users.find(u=>u.email===email&&u.senha===senha);
  if(!user){
    al.textContent='Email ou senha incorretos';al.className='alert error show';return;
  }
  if(user.status==='pending'){
    al.textContent='⏳ Seu acesso ainda está sendo verificado. Aguarde!';al.className='alert info show';return;
  }
  if(user.status==='blocked'){
    al.textContent='🚫 Sua conta está bloqueada. Entre em contato.';al.className='alert error show';return;
  }

  setSession(email);
  loadAreaAluno(user);
  showScreen('area-aluno');
}

}

function loadAdmin(){
  const db=getDB();
  const users=db.users;
  const ativos=users.filter(u=>u.status==='active').length;
  const pendentes=users.filter(u=>u.status==='pending').length;

  document.getElementById('stat-total').textContent=users.length;
  document.getElementById('stat-ativos').textContent=ativos;
  document.getElementById('stat-pendentes').textContent=pendentes;
  document.getElementById('stat-receita').textContent=`R$${(ativos*19.9).toFixed(0)}`;
  document.getElementById('lista-count').textContent=users.length+' alunos';

  // Config
  document.getElementById('admin-pix-key').value=db.config.pixKey||'';
  document.getElementById('admin-whats').value=db.config.whatsapp||'';

  // Lista
  const list=document.getElementById('admin-alunos-list');
  list.innerHTML='';
  if(users.length===0){
    list.innerHTML='<div style="padding:20px;text-align:center;color:#94a3b8;font-size:14px">Nenhum aluno ainda</div>';
    return;
  }
  users.forEach((u,i)=>{
    const row=document.createElement('div');
    row.className='aluno-row';
    const statusClass=u.status==='active'?'status-active':u.status==='pending'?'status-pending':'status-blocked';
    const statusLabel=u.status==='active'?'✅ Ativo':u.status==='pending'?'⏳ Pendente':'🚫 Bloqueado';
    row.innerHTML=`
      <div class="aluno-avatar">👤</div>
      <div class="aluno-row-info">
        <div class="aluno-row-name">${u.nome}</div>
        <div class="aluno-row-email">${u.email} • ${u.whats}</div>
        ${u.comprovante?`<div style="font-size:11px;color:#64748b;margin-top:2px">📎 ${u.comprovante.substring(0,50)}...</div>`:''}
      </div>
      <div style="display:flex;flex-direction:column;gap:4px;align-items:flex-end">
        <span class="aluno-row-status ${statusClass}">${statusLabel}</span>
        <div class="admin-action-btns">
          ${u.status!=='active'?`<button class="admin-btn btn-approve" onclick="aprovarAluno(${i})">✓ Aprovar</button>`:''}
          ${u.status!=='blocked'?`<button class="admin-btn btn-block" onclick="bloquearAluno(${i})">✗ Bloquear</button>`:''}
        </div>
      </div>`;
    list.appendChild(row);
  });
}

function aprovarAluno(idx){

  db.users[idx].status='active';
  db.users[idx].dataAtiv=new Date().toLocaleDateString('pt-BR');
  saveDB(db);
  toast(`✅ ${db.users[idx].nome} aprovado! Avise via WhatsApp.`);
  // Abre WhatsApp automaticamente
  const whats=db.config.whatsapp||'5511999999999';
  const msg=encodeURIComponent(`Olá ${db.users[idx].nome}! 🌾 Seu acesso ao AgroMaster Pro foi liberado! Faça login em: ${window.location.href} com o email: ${db.users[idx].email}`);
  setTimeout(()=>window.open(`https://wa.me/${db.users[idx].whats.replace(/\D/g,'')}?text=${msg}`,'_blank'),800);
  loadAdmin();
}

function bloquearAluno(idx){

}

function init(){
  const session=getSession();
  if(!session) return;

  if(session==='admin'){
    loadAdmin();showScreen('admin');return;
  }

  const db=getDB();
  const user=db.users.find(u=>u.email===session);
  if(user&&user.status==='active'){
    loadAreaAluno(user);showScreen('area-aluno');
  } else {
    clearSession();
