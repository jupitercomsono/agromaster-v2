function showScreen(id){
  document.querySelectorAll('.screen').forEach(s=>s.classList.remove('active'));
  const el = document.getElementById(id);
  if(el) el.classList.add('active');
  const ingame = (id==='playing');
  document.getElementById('hud').style.display = ingame?'flex':'none';
  document.getElementById('ammo-section').style.display = ingame?'block':'none';
  document.getElementById('crosshair').style.display = ingame?'block':'none';
}

function buildAmmoBar(){
  const bar = document.getElementById('ammo-bar');
  bar.innerHTML = '';
  AMMOS_DATA.forEach(a=>{
    const btn = document.createElement('div');
    btn.className = 'ammo-btn' + (a.id===selectedAmmo?' selected':'');
    btn.dataset.id = a.id;
    btn.style.borderColor = a.color;
    btn.style.color = a.color;
    btn.style.background = a.color+'18';
    btn.innerHTML = `<div class="ammo-icon">${a.icon}</div><div class="ammo-name">${a.name}</div><div class="ammo-type">${a.type}</div>`;
    btn.onclick = ()=>selectAmmo(a.id);
    bar.appendChild(btn);
  });
}

function selectAmmo(id){
  selectedAmmo = id;
  document.querySelectorAll('.ammo-btn').forEach(b=>{
    b.classList.toggle('selected', b.dataset.id===id);
  });
  const a = AMMOS_DATA.find(am=>am.id===id);
  if(a) floatNotif(`${a.icon} ${a.name} selecionado`, a.color);
}

// ════════════════════════════
//  SPAWN INIMIGOS
// ════════════════════════════
function getEnemyPool(){
  if(wave<=2) return ['lagarta','percevejo','buva'];
  if(wave<=4) return ['lagarta','percevejo','ferrugem','buva','mosca'];
  return ENEMIES_DATA.map(e=>e.id);
}

function spawnEnemy(){
  if(!gameRunning) return;
  const pool = getEnemyPool();
  const eid = pool[Math.floor(Math.random()*pool.length)];
  const data = ENEMIES_DATA.find(e=>e.id===eid);
  if(!data) return;

  const el = document.createElement('div');
  el.className = 'enemy';
  el.dataset.id = eid;

  const x = 10 + Math.random() * (window.innerWidth - 120);
  el.style.left = x + 'px';
  el.style.top = '-80px';

  el.innerHTML = `
    <div class="enemy-emoji">${data.icon}</div>
    <div class="enemy-name">${data.name}</div>
    <div class="enemy-hp-bar">
      <div class="enemy-hp-fill" style="width:100%;background:${data.color}"></div>
    </div>`;

  el.dataset.hp = data.hp;
  el.dataset.maxHp = data.hp;
  el.dataset.y = -80;
  el.dataset.speed = data.speed * (1 + wave*0.02);

  el.onclick = (e)=>{ e.stopPropagation(); shoot(el); };
  el.addEventListener('touchstart', e=>{ e.preventDefault(); e.stopPropagation(); shoot(el); },{passive:false});

  document.body.appendChild(el);
  enemies.push(el);
}

// ════════════════════════════
//  ATIRAR
// ════════════════════════════
function shoot(el){
  if(!gameRunning || !el || el.classList.contains('dying')) return;

  const eid = el.dataset.id;
  const data = ENEMIES_DATA.find(e=>e.id===eid);
  const ammo = AMMOS_DATA.find(a=>a.id===selectedAmmo);
  if(!data||!ammo) return;

  lastAmmoUsed = ammo;
  lastKilledBy = data;

  const rect = el.getBoundingClientRect();
  const cx = rect.left + rect.width/2;
  const cy = rect.top + rect.height/2;

  if(data.correct.includes(selectedAmmo)){
    // ✅ ACERTO
    let hp = parseInt(el.dataset.hp) - 1;
    el.dataset.hp = hp;
    el.classList.add('hit');
    setTimeout(()=>el.classList.remove('hit'),150);

    // atualiza barra HP
    const fill = el.querySelector('.enemy-hp-fill');
    if(fill) fill.style.width = (hp/parseInt(el.dataset.maxHp)*100)+'%';

    spawnParticles(cx, cy, '💥', ammo.color);

    if(hp<=0){
      // morreu!
      el.classList.add('dying');
      kills++;
      const pts = data.pts + wave*10;
      score += pts;
      floatNotif(`+${pts} ☠️ ${data.name} eliminada!`, '#f9c74f', cy);
      flashGood();
      setTimeout(()=>removeEnemy(el), 400);
    } else {
      floatNotif(`💥 ${ammo.name} funcionou! HP: ${hp}`, ammo.color, cy);
      flashGood();
    }
  } else {
    // ❌ ERRO — perde vida ao aplicar errado!
    spawnParticles(cx, cy, '🚫', '#ef4444');
    flashHit();
    lives--;
    updateHUD();
    const wrongMsg = typeof data.wrongMsg==='function'?data.wrongMsg(ammo):data.wrongMsg;
    floatNotif(`❌ ${ammo.name} ERRADO p/ ${data.name}! -1 vida`, '#ef4444', cy);
    lastAmmoUsed = ammo;
    lastKilledBy = data;
    if(lives<=0){
      setTimeout(()=>triggerGameOver(),800);
    }
  }
  updateHUD();
}

// ════════════════════════════
//  UPDATE LOOP
// ════════════════════════════
function update(){
  if(!gameRunning) return;

  enemies.forEach((el,i)=>{
    if(!el.parentNode) return;
    if(el.classList.contains('dying')) return;

    let y = parseFloat(el.dataset.y) + parseFloat(el.dataset.speed);
    el.dataset.y = y;
    el.style.top = y + 'px';

    if(y > window.innerHeight){
      // inimigo passou → só remove, não perde vida
      removeEnemy(el);
    }
  });

  // wave: aumenta dificuldade por kills
  if(kills > 0 && kills % 8 === 0){
    wave = Math.floor(kills/8) + 1;
    spawnDelay = Math.max(600, 2000 - wave*150);
    updateHUD();
  }

  requestAnimationFrame(update);
}

function removeEnemy(el){
  const i = enemies.indexOf(el);
  if(i>-1) enemies.splice(i,1);
  if(el.parentNode) el.remove();
}

// ════════════════════════════
//  VIDAS
// ════════════════════════════
function loseLife(data){
  lives--;
  flashHit();
  updateHUD();
  if(data) floatNotif(`💔 ${data.name} passou! -1 vida`, '#ef4444');
  if(lives<=0) triggerGameOver();
}

function triggerGameOver(){
  gameRunning = false;
  clearInterval(spawnInterval);

  const data = lastKilledBy || ENEMIES_DATA[0];
  const ammo = lastAmmoUsed || AMMOS_DATA[0];
  const correctAmmosText = data.correct.map(aid=>{
    const a = AMMOS_DATA.find(am=>am.id===aid);
    return a?`${a.icon} ${a.name}`:aid;
  }).join(', ');

  document.getElementById('go-enemy').textContent = data.icon;
  document.getElementById('go-killed-by').textContent = `Derrotado por: ${data.name}`;
  document.getElementById('go-wrong').innerHTML =
    `Você usou <b>${ammo.icon} ${ammo.name}</b> (${ammo.type})<br><br>${typeof data.wrongMsg==='function'?data.wrongMsg(ammo):data.wrongMsg}`;
  document.getElementById('go-right').innerHTML =
    `Corretos: <b>${correctAmmosText}</b><br><br>${data.rightMsg}`;
  document.getElementById('go-pts').textContent = score;
  document.getElementById('go-kills').textContent = kills;
  document.getElementById('go-wave').textContent = wave;

  // limpa inimigos
  enemies.forEach(e=>{if(e.parentNode)e.remove();});
  enemies=[];

  showScreen('gameover');
}

function updateHUD(){
  document.getElementById('score-display').textContent = `⭐ ${score}`;
  document.getElementById('wave-display').textContent = `WAVE ${wave}`;
  document.getElementById('lives-display').textContent =
    '❤️'.repeat(Math.max(0,lives))+'🖤'.repeat(Math.max(0,3-lives));
}

// ════════════════════════════
//  EFEITOS VISUAIS
// ════════════════════════════
function flashHit(){
  const el=document.getElementById('hit-flash');
  el.classList.remove('show');void el.offsetWidth;el.classList.add('show');
}
function flashGood(){
  const el=document.getElementById('good-flash');
  el.classList.remove('show');void el.offsetWidth;el.classList.add('show');
}

function spawnParticles(x,y,emoji,color){
  for(let i=0;i<5;i++){
    const p=document.createElement('div');
    p.className='particle';
    p.textContent=emoji;
    p.style.left=x+'px';p.style.top=y+'px';
    p.style.setProperty('--dx',(Math.random()-0.5)*80+'px');
    p.style.setProperty('--dy',(Math.random()-1)*60+'px');
    p.style.color=color;
    document.body.appendChild(p);
    setTimeout(()=>p.remove(),600);
  }
}

function floatNotif(msg, color='#fff', y=null){
  const el=document.createElement('div');
  el.className='float-notif';
  el.textContent=msg;el.style.color=color;
  if(y) el.style.top=Math.min(y,window.innerHeight*.7)+'px';
  document.body.appendChild(el);
  setTimeout(()=>el.remove(),800);
}

//  INICIAR
// ════════════════════════════
function startGame(){
  score=0;lives=3;kills=0;wave=1;spawnDelay=2000;
  lastKilledBy=null;lastAmmoUsed=null;
  enemies.forEach(e=>{if(e.parentNode)e.remove();});
  enemies=[];
  gameRunning=true;
  buildAmmoBar();
  updateHUD();
  showScreen('playing');
  clearInterval(spawnInterval);
  spawnInterval=setInterval(()=>{
    spawnEnemy();
    clearInterval(spawnInterval);
    spawnInterval=setInterval(spawnEnemy, spawnDelay);
  },300);
  requestAnimationFrame(update);
}
